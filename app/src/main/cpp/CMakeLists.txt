cmake_minimum_required(VERSION 3.22.1)
project(voxstudio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Optimize C kernels too: Gradle's Debug build otherwise leaves GGML C loops at O0.
set(CMAKE_C_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
add_compile_options(-O3)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
option(VOX_MOBILE_FAST "FP16/dot-product ARM CPU build, guarded at runtime" OFF)
if(VOX_MOBILE_FAST)
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
else()
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
# Probe remains baseline ARMv8 even when the math backend uses newer instructions.
# The compatible engine is built by :engine-compat from these same sources, without the probe.
option(VOX_BUILD_PROBE "Build the CPU capability probe" ON)
if(VOX_BUILD_PROBE)
    add_library(voxprobe SHARED capabilities.cpp)
    target_link_options(voxprobe PRIVATE "-Wl,-z,max-page-size=16384")
endif()
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor)
set(VOX ${VENDOR}/voxcpm)
option(VOX_MOBILE_OPENCL "Experimental Adreno OpenCL" OFF)
if(VOX_MOBILE_OPENCL)
    set(Python3_EXECUTABLE "/usr/bin/python3" CACHE FILEPATH "" FORCE)
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    set(OpenCL_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../dev-deps/OpenCL-Headers-2024.10.24" CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../dev-deps/libOpenCL.so" CACHE FILEPATH "" FORCE)
    add_compile_definitions(VOX_MOBILE_OPENCL=1)
else()
    set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
endif()
option(VOX_MOBILE_GPU "Experimental Vulkan backend" OFF)
if(VOX_MOBILE_GPU)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    set(Vulkan_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../dev-deps/vulkan/Vulkan-Headers-1.4.321/include" CACHE PATH "" FORCE)
    set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/darwin-x86_64/glslc" CACHE FILEPATH "" FORCE)
    add_compile_definitions(VOX_MOBILE_GPU=1)
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(${VOX}/third_party/ggml ${CMAKE_BINARY_DIR}/ggml)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/json/nlohmann)
configure_file(${VENDOR}/json.hpp ${CMAKE_BINARY_DIR}/json/nlohmann/json.hpp COPYONLY)
# Add Android progress hooks to a build-local copy of the pinned CLI adapter.
# Upstream inference implementations and their ownership contracts are unchanged.
file(READ ${VOX}/examples/voxcpm_tts.cpp MOBILE_PIPELINE)
# Persistent model session (mobile_session.inc): splice the per-call load block out of main.
set(SESSION_START "VoxCPMBackend backend(options.backend, options.threads);")
set(SESSION_END "ChineseCharSplitTokenizer split_tokenizer(tokenizer);")
string(FIND "${MOBILE_PIPELINE}" "${SESSION_START}" SESSION_START_AT)
string(FIND "${MOBILE_PIPELINE}" "${SESSION_END}" SESSION_END_AT)
if(SESSION_START_AT EQUAL -1 OR SESSION_END_AT EQUAL -1 OR SESSION_END_AT LESS SESSION_START_AT)
    message(FATAL_ERROR "Pinned voxcpm_tts.cpp changed: update the mobile session splice")
endif()
string(LENGTH "${SESSION_END}" SESSION_END_LENGTH)
math(EXPR SESSION_TAIL_AT "${SESSION_END_AT} + ${SESSION_END_LENGTH}")
string(SUBSTRING "${MOBILE_PIPELINE}" 0 ${SESSION_START_AT} SESSION_HEAD)
string(SUBSTRING "${MOBILE_PIPELINE}" ${SESSION_TAIL_AT} -1 SESSION_TAIL)
set(MOBILE_PIPELINE "${SESSION_HEAD}MobileSession& mobile = mobile_session(options); VoxCPMBackend& backend = *mobile.backend; auto& store = mobile.store; VoxCPMRuntime& runtime = mobile.runtime; AudioVAE& audio_vae = mobile.audio_vae; ChineseCharSplitTokenizer& split_tokenizer = *mobile.split_tokenizer;${SESSION_TAIL}")
foreach(MARKER "PreparedInputs prepare_inputs(const Options& options," "prepared.prompt_feat = load_audio_features(options.prompt_audio_path,")
    string(FIND "${MOBILE_PIPELINE}" "${MARKER}" MARKER_AT)
    if(MARKER_AT EQUAL -1)
        message(FATAL_ERROR "Pinned voxcpm_tts.cpp changed: missing ${MARKER}")
    endif()
endforeach()
string(REPLACE "PreparedInputs prepare_inputs(const Options& options," "\n#include \"mobile_session.inc\"\nPreparedInputs prepare_inputs(const Options& options," MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "prepared.prompt_feat = load_audio_features(options.prompt_audio_path," "prepared.prompt_feat = mobile_cached_audio_features(options.prompt_audio_path," MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "const Options options = parse_args(argc, argv);" "const Options options = parse_args(argc, argv); mobile_progress(\"Loading model\", 0, 0);" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "const PreparedInputs prepared = prepare_inputs(" "mobile_progress(options.prompt_audio_path.empty() ? \"Preparing script\" : \"Encoding voice reference\", 0, 0); mobile_set_threads(backend, mobile_vae_threads(options.threads)); const PreparedInputs prepared = prepare_inputs(" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "VoxCPMDecodeState state = runtime.prefill(" "mobile_progress(\"Preparing speech\", 0, 0); mobile_set_threads(backend, options.threads); VoxCPMDecodeState state = runtime.prefill(" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "for (int step = 0; step < max_len; ++step) {" "for (int step = 0; step < max_len; ++step) { mobile_check_cancel();" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "const auto decode_start = std::chrono::steady_clock::now();" "mobile_progress(\"Decoding waveform\", 0, 0); mobile_set_threads(backend, mobile_vae_threads(options.threads)); const auto decode_start = std::chrono::steady_clock::now();" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "state = std::move(result.output_1);" "state = std::move(result.output_1); mobile_progress(\"Generating speech\", step + 1, max_len, static_cast<double>(step + 1) * decode_patch_len / audio_vae.config().output_sample_rate());" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
string(REPLACE "std::mt19937 rng(std::random_device{}());" "std::mt19937 rng(mobile_seed());" MOBILE_PIPELINE "${MOBILE_PIPELINE}")
file(WRITE ${CMAKE_BINARY_DIR}/mobile_pipeline.cpp "${MOBILE_PIPELINE}")
file(GLOB CORE ${VOX}/src/*.cpp)
list(FILTER CORE EXCLUDE REGEX "/server_common.cpp$")
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/bridge.cpp ENGINE_BRIDGE)
if(VOX_MOBILE_OPENCL OR VOX_MOBILE_GPU)
    string(REPLACE "Java_com_voxstudio_app_NativeEngine_" "Java_com_voxstudio_app_NativeGpuEngine_" ENGINE_BRIDGE "${ENGINE_BRIDGE}")
    set(ENGINE_NAME voxstudio_gpu)
else()
    string(REPLACE "Java_com_voxstudio_app_NativeEngine_" "Java_com_voxstudio_app_NativeCpuEngine_" ENGINE_BRIDGE "${ENGINE_BRIDGE}")
    # Both CPU engines expose the same JNI names; the app loads exactly one of them.
    if(VOX_MOBILE_FAST)
        set(ENGINE_NAME voxstudio_cpu_fast)
    else()
        set(ENGINE_NAME voxstudio_cpu)
    endif()
endif()
file(WRITE ${CMAKE_BINARY_DIR}/engine_bridge.cpp "${ENGINE_BRIDGE}")
add_library(voxstudio SHARED ${CMAKE_BINARY_DIR}/engine_bridge.cpp ${CORE} ${VOX}/third_party/miniaudio/miniaudio.c)
target_include_directories(voxstudio PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${VOX}/include ${VOX}/third_party/miniaudio ${VOX}/third_party/ggml/include ${VOX}/third_party/ggml/src ${CMAKE_BINARY_DIR}/json ${CMAKE_BINARY_DIR})
target_compile_definitions(voxstudio PRIVATE VOXCPM_ENABLE_MP3=0 VOXCPM_ENABLE_OPUS=0 MA_NO_DEVICE_IO)
target_link_libraries(voxstudio PRIVATE ggml ggml-base android log)
target_link_options(voxstudio PRIVATE "-Wl,-z,max-page-size=16384")

set_target_properties(voxstudio PROPERTIES OUTPUT_NAME ${ENGINE_NAME})
